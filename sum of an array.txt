 #include<stdio.h>
 int main()
  {
   int ar[10],i,n,sum=0;
    printf("enter size of an array:");
     scanf("%d", &n);
     printf("enter elements:");
    for(i=1;i<=n;i++)
    {
        scanf("%d", &ar[i]);
    }
    for(i=1;i<=n;i++){
        sum=sum+ar[i];
    }
    printf("sum of an array is %d\n",sum);
    return 0;
  }
