#include <stdio.h>

int main() {
    int ar[10],c1=0,c2=0,n,i;
    printf("enter size of an array:");
    scanf("%d",&n);
    printf("enter elements:");
    for(i=1;i<=n;i++)
    {
        scanf("%d",&ar[i]);
    }
    for(i=1;i<=n;i++)
    {
    if(ar[i]%2==0){
        c1++;
    }else{
        c2++;
    }
  }
    printf("even numbers are: %d \n",c1);
    printf("odd numbers are: %d \n",c2);
    return 0;
}
